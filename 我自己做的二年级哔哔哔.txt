-- 整体拖拽逻辑
local dragging = false
local dragStartPos
local frameStartPos
mainFrame.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 then
		dragging = true
		dragStartPos = input.Position
		frameStartPos = mainFrame.Position
	end
end)
UserInputService.InputChanged:Connect(function(input)
	if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
		local delta = input.Position - dragStartPos
		mainFrame.Position = UDim2.new(frameStartPos.X.Scale,frameStartPos.X.Offset+delta.X,frameStartPos.Y.Scale,frameStartPos.Y.Offset+delta.Y)
	end
end)
UserInputService.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 then dragging = false end
end)

-- 按钮功能
btnFly.MouseButton1Click:Connect(function()
	flyActive = true
	local char = plr.Character or plr.CharacterAdded:Wait()
	local hum = char:WaitForChild("Humanoid")
	hum.GravityScale = 0
	hum:SetStateEnabled(Enum.HumanoidStateType.Jumping,false)
end)

btnAdd.MouseButton1Click:Connect(function()
	flySpeed +=1
	speedLabel.Text = tostring(flySpeed)
end)

btnSub.MouseButton1Click:Connect(function()
	if flySpeed>1 then
		flySpeed -=1
		speedLabel.Text = tostring(flySpeed)
	end
end)

	mainFrame.Visible = false

btnStopFly.MouseButton1Click:Connect(function()
	flyActive = false
	local char = plr.Character
	if char then
		local hum = char:FindFirstChild("Humanoid")
		if hum then
			hum.GravityScale =1
			hum:SetStateEnabled(Enum.HumanoidStateType.Jumping,true)
		end
	end
end)

-- 核心飞行逻辑：摇杆控制方向，人物跟随视角转向，高度固定
RunService.RenderStepped:Connect(function(dt)
	if not flyActive then return end
	local char = plr.Character
	if not char then return end
	local root = char:FindFirstChild("HumanoidRootPart")
	local hum = char:FindFirstChild("Humanoid")
	if not root or not hum then return end
	
	local cam = workspace.CurrentCamera
	-- 获取相机水平朝向
	local camLook = cam.CFrame.LookVector
	local camRight = cam.CFrame.RightVector
	camLook = Vector3.new(camLook.X, 0, camLook.Z).Unit
	camRight = Vector3.new(camRight.X, 0, camRight.Z).Unit
	
	-- 获取移动输入（摇杆/WASD）
	local moveDir = hum.MoveDirection
	if moveDir.Magnitude > 0.01 then
		-- 基于相机朝向计算世界移动方向
		local worldMove = (camLook * moveDir.Z + camRight * moveDir.X).Unit
		root.CFrame += worldMove * flySpeed * dt
	end
	
	-- 人物朝向跟随相机水平方向
	local targetYaw = math.atan2(camLook.X, camLook.Z)
	root.CFrame = CFrame.new(root.Position) * CFrame.Angles(0, targetYaw, 0)
end)
